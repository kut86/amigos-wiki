import { loginWithGoogle, onAuthChange, ensureUser, getUserRole } from "./firebase.js"; /* ───────────────────────── DOM ───────────────────────── */ const loginBtn = document.getElementById("loginBtn"); const statusText = document.getElementById("status"); // можно добавить в HTML /* ───────────────────────── LOGIN CLICK ───────────────────────── */ loginBtn.addEventListener("click", async () => { try { statusText && (statusText.textContent = "Вход..."); await loginWithGoogle(); } catch (err) { console.error("LOGIN ERROR:", err); statusText && (statusText.textContent = "Ошибка входа"); } }); /* ───────────────────────── AUTH LISTENER (ГЛАВНАЯ ЛОГИКА) ───────────────────────── */ onAuthChange(async (user) => { if (!user) { statusText && (statusText.textContent = "Не авторизован"); return; } statusText && (statusText.textContent = "Проверка пользователя..."); try { // создаём пользователя если его нет await ensureUser(user); // получаем роль const role = await getUserRole(user.uid); // сохраняем локально (временно, для map/wiki) localStorage.setItem("role", role); localStorage.setItem("uid", user.uid); console.log("[AUTH] role:", role); statusText && (statusText.textContent = "Успешный вход"); /* ───────────────────────── REDIRECT LOGIC ───────────────────────── */ setTimeout(() => { if (role === "admin") { window.location.href = "map.html"; } else { window.location.href = "map.html"; } }, 500); } catch (err) { console.error("AUTH ERROR:", err); statusText && (statusText.textContent = "Ошибка авторизации"); } });
